** enum + pattern matching
#+begin_src rust
enum Status {
    Ok,                    // вариант без данных
    Error,                 // вариант без данных
    Timeout(u64),          // вариант С данными (время ожидания)
    Custom(String),        // вариант со строкой
}

fn main() {
    let status1 = Status::Ok;
    let status2 = Status::Timeout(5000);    // хранит число 5000
    let status3 = Status::Custom("Failed".to_string());

    // Pattern matching для доступа к данным:
    match status2 {
        Status::Ok => println!("Успех"),
        Status::Error => println!("Ошибка"),
        Status::Timeout(ms) => println!("Таймаут через {} мс", ms),
        Status::Custom(msg) => println!("Ошибка: {}", msg),
    }
}

** Обработки result, option :rust:result:option:
#+begin_src rust
enum Option<T> {
    Some(T),  // Значение есть
    None,     // Значения нет
}
fn find_even(numbers: &[i32]) -> Option<&i32> {
    for num in numbers {
        if num % 2 == 0 {
            return Some(num);
        }
    }
    None
}

enum Result<T, E> {
    Ok(T),   // Успех с результатом типа T
    Err(E),  // Ошибка типа E
}
fn divide(dividend: f64, divisor: f64) -> Result<f64, String> {
    if divisor == 0.0 {
        Err(String::from("Деление на ноль!"))
    } else {
        Ok(dividend / divisor)
    }
}

fn main() {
    let result = divide(10.0, 0.0);

    match result {
        Ok(value) => println!("Результат: {}", value),
        Err(error) => println!("Ошибка: {}", error),
    }
}

** Форматы вывода
| Формат | Трейт    | Назначение                          | Пример вывода                     |
| {}     | Display  | Для пользователей, красивый вывод   | 42 или Hello                      |
| {:?}   | Debug    | Для разработчиков, отладка          | Person { name: "Alice", age: 30 } |
| {:#?}  | Debug    | "Pretty" Debug (с отступами)        | Многострочный        |
| {:b}   | Binary   | Двоичный формат                     | 101010                            |
| {:x}   | LowerHex | Шестнадцатеричный (нижний регистр)  | deadbeef                          |
| {:X}   | UpperHex | Шестнадцатеричный (верхний регистр) | DEADBEEF                          |
| {:o}   | Octal    | Восьмеричный формат                 | 755                               |
| {:e}   | LowerExp | Экспоненциальный (нижний регистр)   | 1.23e4                            |
